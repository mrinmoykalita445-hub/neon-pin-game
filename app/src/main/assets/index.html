  function loadPreferences() {

    try {

      const saved = JSON.parse(localStorage.getItem(STORAGE_KEY) || "{}");

      best = Number.isFinite(saved.best) ? Math.max(0, Math.floor(saved.best)) : 0;

      muted = saved.muted === true;

    } catch (_) {}

  }

    try {

      localStorage.setItem(STORAGE_KEY, JSON.stringify({ best, muted }));

    } catch (_) {

      // Private mode or denied storage must never interrupt a game.

    }

  }

  function normalize(a) {

    return ((a % TAU) + TAU) % TAU;

  }

  function angleDifference(a, b) {

    return Math.atan2(Math.sin(a - b), Math.cos(a - b));

  }

  function clamp(value, min, max) {

    return Math.max(min, Math.min(max, value));

  }

  function random(min, max) {

    return min + Math.random() * (max - min);

  }

  function unlockAudio() {

    if (muted) return;

    try {

      if (!audio) {

        const AudioContextClass = window.AudioContext || window.webkitAudioContext;

        if (!AudioContextClass) return;

        audio = new AudioContextClass();

        master = audio.createGain();

        master.gain.value = 0.5;

        const limiter = audio.createDynamicsCompressor();

        limiter.threshold.value = -15;

        limiter.knee.value = 15;

        limiter.ratio.value = 8;

        master.connect(limiter);

        limiter.connect(audio.destination);

        noiseBuffer = audio.createBuffer(

          1, Math.floor(audio.sampleRate * 0.35), audio.sampleRate

        );

        const data = noiseBuffer.getChannelData(0);

        for (let i = 0; i < data.length; i++) {

          data[i] = Math.random() * 2 - 1;

        }

      }

      if (audio.state === "suspended") audio.resume().catch(() => {});

    } catch (_) {

      // Sound is optional; unsupported audio never blocks gameplay.

    }

  }

  function tone(freq, endFreq, duration, type, volume, delay = 0) {

    if (!audio || !master || muted || audio.state !== "running") return;

    const start = audio.currentTime + delay;

    const oscillator = audio.createOscillator();

    const gain = audio.createGain();

    oscillator.type = type;

    oscillator.frequency.setValueAtTime(Math.max(1, freq), start);

    oscillator.frequency.exponentialRampToValueAtTime(

      Math.max(1, endFreq), start + duration

    );

    gain.gain.setValueAtTime(0.0001, start);

    gain.gain.exponentialRampToValueAtTime(volume, start + 0.006);

    gain.gain.exponentialRampToValueAtTime(0.0001, start + duration);

    oscillator.connect(gain);

    gain.connect(master);

    oscillator.start(start);

    oscillator.stop(start + duration + 0.02);

    oscillator.onended = () => {

      oscillator.disconnect();

      gain.disconnect();

    };

  }

  function noise(duration, volume, frequency) {

    if (!audio || !noiseBuffer || muted || audio.state !== "running") return;

    const source = audio.createBufferSource();

    const filter = audio.createBiquadFilter();

    const gain = audio.createGain();

    const now = audio.currentTime;

    source.buffer = noiseBuffer;

    filter.type = "bandpass";

    filter.frequency.value = frequency;

    filter.Q.value = 0.8;

    gain.gain.setValueAtTime(volume, now);

    gain.gain.exponentialRampToValueAtTime(0.0001, now + duration);

    source.connect(filter);

    filter.connect(gain);

    gain.connect(master);

    source.start(now);

    source.stop(now + duration);

    source.onended = () => {

      source.disconnect();

      filter.disconnect();

      gain.disconnect();

    };

  }

  function playSound(kind) {

    if (muted || !audio) return;

    try {

      if (kind === "launch") {

        tone(380, 920, 0.065, "sine", 0.10);

        noise(0.045, 0.07, 1900);

      } else if (kind === "hit") {

        tone(150, 65, 0.12, "sine", 0.45);

        tone(1150, 630, 0.08, "triangle", 0.11);

        noise(0.055, 0.15, 1300);

      } else if (kind === "fail") {

        tone(195, 42, 0.32, "sawtooth", 0.20);

        tone(710, 120, 0.23, "square", 0.10);

        noise(0.26, 0.29, 2500);

      } else if (kind === "clear") {

        [523.25, 659.25, 783.99, 1046.5].forEach((f, i) => {

          tone(f, f * 1.003, 0.32, "sine", 0.24, i * 0.085);

          tone(f * 2, f * 2, 0.19, "triangle", 0.045, i * 0.085);

        });

      }

    } catch (_) {}

  }

    height = Math.max(1, rect.height);

    dpr = Math.max(1, window.devicePixelRatio || 1);

    canvas.width = Math.round(width * dpr);

    canvas.height = Math.round(height * dpr);

cx = width * 0.5;

    cy = height * 0.36;

    radius = Math.max(10, Math.min(width * 0.205, height * 0.20, 88));

    pinLength = clamp(height * 0.105, 15, 40);

    launchY = Math.max(cy + radius + pinLength + 12, height - pinLength - 14);

    flightSpeed = Math.max(850, height * 2.6);

    if (projectile) {

      const oldDistance = oldLaunchY - (oldCy + oldRadius);

      const progress = oldDistance > 0

        ? clamp((oldLaunchY - projectile.y) / oldDistance, 0, 1) : 0;

      projectile.y = launchY - progress * (launchY - cy - radius);

    }

    if (oldWidth > 1 && oldHeight > 1) {

      for (const p of particles) {

        p.x *= width / oldWidth;

        p.y *= height / oldHeight;

      }

      for (const p of popups) {

        p.x *= width / oldWidth;

        p.y *= height / oldHeight;

      }

    }

    if (brokenPin) {

      brokenPin.x = cx;

      brokenPin.y = cy + radius + pinLength;

    }

  }

  function motionAt(time) {

    const base = Math.min(3.35, 1.0 + (level - 1) * 0.105);

    const direction = level % 2 ? 1 : -1;

    const pattern = (level - 1) % 4;

    if (pattern === 0) {

      return direction * base * (1 + 0.17 * Math.sin(time * 1.7));

    }

    if (pattern === 1) {

      return direction * base *

        (0.92 + 0.55 * Math.sin(time * 1.65 + 0.5));

    }

    if (pattern === 2) {

      // Continuous reversals, with an actual zero-speed crossing.

      return direction * base * 1.3 * Math.cos(time * 0.92);

    }

    // Brief, deterministic pauses with soft acceleration around the stop.

    const cycle = (time + 0.7) % 3.9;

    let envelope = 1;

    if (cycle >= 2.70 && cycle < 2.88) {

      envelope = 1 - (cycle - 2.70) / 0.18;

    } else if (cycle >= 2.88 && cycle < 3.25) {

      envelope = 0;

    } else if (cycle >= 3.25 && cycle < 3.47) {

      envelope = (cycle - 3.25) / 0.22;

    }

    return direction * base * envelope *

      (1 + 0.18 * Math.sin(time * 2));

  }

  function configureLevel() {

    needed = Math.min(18, 6 + (level - 1) * 2);

    remaining = needed;

    levelTime = 0;

    angle = normalize(level * 0.73);

    projectile = null;

    brokenPin = null;

    pins = [];

    hitFlash = 0;

    // Gold pins are pre-existing obstacles, not shots still to be launched.

    const obstacles = Math.min(7, Math.floor(level / 2));

    for (let i = 0; i < obstacles; i++) {

      pins.push({

        localAngle: normalize((i / obstacles) * TAU + 0.23),

        color: "#ffd700",

        obstacle: true

      });

    }

    const patterns = [

      "STEADY ORBIT",

      "PULSE · VARIABLE SPEED",

      "SWITCHBACK · REVERSING",

      "SUSPENSE · BRIEF PAUSES"

    ];

    $("mode-label").textContent = patterns[(level - 1) % 4];

    $("toast").classList.remove("visible");

    updateHUD();

  }

  function startGame() {

    unlockAudio();

    score = 0;

    level = 1;

    hasNewBest = false;

    particles = [];

    popups = [];

    shake = 0;

    state = "playing";

    stateTime = 0;

    accumulator = 0;

    configureLevel();

    hideModal();

  }

  function nextLevel() {

    level++;

    state = "playing";

    stateTime = 0;

    configureLevel();

  }

  function launch() {

    if (state !== "playing" || projectile || remaining <= 0) return;

    projectile = { y: launchY };

    playSound("launch");

    updateHUD();

  }

  function burst(x, y, count, palette = COLORS, strength = 1) {

    const actual = reducedMotion ? Math.ceil(count * 0.25) : count;

    for (let i = 0; i < actual; i++) {

      const a = random(0, TAU);

      const speed = random(40, 260) * strength;

      const life = random(0.35, 1.05);

      particles.push({

        x, y,

        vx: Math.cos(a) * speed,

        vy: Math.sin(a) * speed,

        life, maxLife: life,

        size: random(1.4, 4.1),

        color: palette[Math.floor(Math.random() * palette.length)]

      });

    }

    if (particles.length > 380) particles.splice(0, particles.length - 380);

  }

  function successfulHit() {

    const impactAngle = Math.atan2((cy + radius) - cy, cx - cx);

    const localAngle = normalize(impactAngle - angle);

    pins.push({ localAngle, color: "#00f2fe", obstacle: false });

    projectile = null;

    remaining--;

    score += 10;

    hitFlash = 1;

    if (score > best) {

      best = score;

      hasNewBest = true;

      savePreferences();

    }

    playSound("hit");

    burst(cx, cy + radius, 10, ["#00f2fe", "#b6ffff"], 0.45);

    popups.push({

      x: cx + 22, y: cy + radius + 13,

      text: "+10", life: 0.65, maxLife: 0.65, color: "#7affed"

    });

    pulseScore();

    if (remaining === 0) {

      state = "clearing";

      stateTime = 0;

      playSound("clear");

      burst(cx, cy, 100);

      $("toast").classList.add("visible");

    }

    updateHUD();

  }

  function gameOver() {

    if (state !== "playing") return;

    const y = projectile ? projectile.y : cy + radius;

    brokenPin = {

      x: cx,

      y,

      vx: random(-145, 145),

      vy: -110,

      rotation: 0,

      spin: random(-7, 7)

    };

    projectile = null;

    state = "dying";

    stateTime = 0;

    shake = reducedMotion ? 0 : 1;

    hitFlash = 0;

    playSound("fail");

    burst(cx, y, 58, ["#ff577f", "#ffd700", "#c88cff", "#ffffff"]);

    savePreferences();

    updateHUD();

  }

  function pointSegmentDistanceSquared(px, py, ax, ay, bx, by) {

    const dx = bx - ax;

    const dy = by - ay;

    const lengthSquared = dx * dx + dy * dy;

    const t = lengthSquared > 0

      ? clamp(((px - ax) * dx + (py - ay) * dy) / lengthSquared, 0, 1)

      : 0;

    const x = ax + dx * t;

    const y = ay + dy * t;

    return (px - x) ** 2 + (py - y) ** 2;

  }

  function cross(ax, ay, bx, by) {

    return ax * by - ay * bx;

  }

  function segmentDistanceSquared(ax, ay, bx, by, cx2, cy2, dx, dy) {

    const rx = bx - ax, ry = by - ay;

    const sx = dx - cx2, sy = dy - cy2;

    const denominator = cross(rx, ry, sx, sy);

    if (Math.abs(denominator) > 1e-10) {

      const qx = cx2 - ax, qy = cy2 - ay;

      const t = cross(qx, qy, sx, sy) / denominator;

      const u = cross(qx, qy, rx, ry) / denominator;

      if (t >= 0 && t <= 1 && u >= 0 && u <= 1) return 0;

    }

    return Math.min(

      pointSegmentDistanceSquared(ax, ay, cx2, cy2, dx, dy),

      pointSegmentDistanceSquared(bx, by, cx2, cy2, dx, dy),

      pointSegmentDistanceSquared(cx2, cy2, ax, ay, bx, by),

      pointSegmentDistanceSquared(dx, dy, ax, ay, bx, by)

    );

  }

  function checkCollision(tipY) {

    // Capsule hitboxes cover the complete exposed shafts, not just

    // the final attachment angle. This catches outer-shaft clashes too.

    const collisionDistance = PIN_RADIUS * 2;

    const thresholdSquared = collisionDistance * collisionDistance;

    // Pins can only intersect the central launch corridor close to bottom.

    const angularWindow =

      Math.asin(Math.min(1, collisionDistance / radius)) + 0.03;

    for (const pin of pins) {

      const worldAngle = pin.localAngle + angle;

      const difference = angleDifference(worldAngle, BOTTOM);

      if (Math.abs(difference) > angularWindow) continue;

      const ux = Math.cos(worldAngle);

      const uy = Math.sin(worldAngle);

      const ax = cx + ux * radius;

      const ay = cy + uy * radius;

      const bx = cx + ux * (radius + pinLength);

      const by = cy + uy * (radius + pinLength);

      if (segmentDistanceSquared(

        cx, tipY, cx, tipY + pinLength, ax, ay, bx, by

      ) <= thresholdSquared) {

        return true;

      }

    }

    return false;

  }

  let pausedFrom = "playing";

  function pauseGame() {

    if (!["playing", "clearing"].includes(state)) return;

    pausedFrom = state;

    state = "paused";

    showModal("paused");

    updateHUD();

  }

  function resumeGame() {

    if (state !== "paused") return;

    unlockAudio();

    state = pausedFrom;

    accumulator = 0;

    hideModal();

    updateHUD();

  }

  function primaryAction() {

    unlockAudio();

    if (state === "ready" || state === "over") startGame();

    else if (state === "paused") resumeGame();

    else if (state === "playing") launch();

  }

  function stepPlaying(dt) {

    // Adaptive microsteps keep both projectile travel and rotating-pin

    // travel below 1.35 CSS pixels per collision sample.

    const maxWheelSpeed = 6.8;

    const maxMotion = Math.max(

      projectile ? flightSpeed : 0,

      maxWheelSpeed * (radius + pinLength)

    );

    const steps = Math.max(1, Math.ceil(maxMotion * dt / 1.35));

    const micro = dt / steps;

    for (let i = 0; i < steps && state === "playing"; i++) {

      rotationSpeed = motionAt(levelTime + micro * 0.5);

      angle = normalize(angle + rotationSpeed * micro);

      levelTime += micro;

      if (!projectile) continue;

      const targetY = cy + radius;

      const nextY = Math.max(targetY, projectile.y - flightSpeed * micro);

      if (checkCollision(nextY)) {

        projectile.y = nextY;

        gameOver();

        break;

      }

      projectile.y = nextY;

      if (nextY <= targetY + 0.00001) successfulHit();

    }

  }

  function update(dt) {

    if (state === "paused") return;

    visualTime += dt;

    stateTime += dt;

    hitFlash = Math.max(0, hitFlash - dt * 4.5);

    shake = Math.max(0, shake - dt * 2.6);

    if (state === "playing") {

      stepPlaying(dt);

    } else if (state === "ready") {

      angle = normalize(angle + dt * 0.38);

    } else if (state === "clearing") {

      angle = normalize(angle + dt * rotationSpeed * 0.28);

      if (stateTime > 1.35) nextLevel();

    } else if (state === "dying") {

      if (brokenPin) {

        brokenPin.x += brokenPin.vx * dt;

        brokenPin.y += brokenPin.vy * dt;

        brokenPin.vy += 720 * dt;

        brokenPin.rotation += brokenPin.spin * dt;

      }

      if (stateTime > 0.48) {

        state = "over";

        stateTime = 0;

        showModal("over");

        updateHUD();

      }

    }

    for (let i = particles.length - 1; i >= 0; i--) {

      const p = particles[i];

      p.life -= dt;

      if (p.life <= 0) {

        particles.splice(i, 1);

        continue;

      }

      p.x += p.vx * dt;

      p.y += p.vy * dt;

      p.vx *= Math.exp(-1.8 * dt);

      p.vy += 135 * dt;

    }

    for (let i = popups.length - 1; i >= 0; i--) {

      const p = popups[i];

      p.life -= dt;

      p.y -= 42 * dt;

      if (p.life <= 0) popups.splice(i, 1);

    }

  }

  function drawPin(x, y, direction, color, alpha = 1) {

    // Direction points outward from the wheel. The pointed end is at x,y.

    const ux = Math.cos(direction), uy = Math.sin(direction);

    const nx = -uy, ny = ux;

    const taper = Math.min(8, pinLength * 0.3);

    const halfWidth = PIN_RADIUS * 0.66;

    ctx.save();

    ctx.globalAlpha = alpha;

    ctx.fillStyle = color;

    ctx.shadowColor = color;

    ctx.shadowBlur = 9;

    ctx.beginPath();

    ctx.moveTo(x, y);

    ctx.lineTo(x + ux * taper + nx * halfWidth,

               y + uy * taper + ny * halfWidth);

    ctx.lineTo(x + ux * pinLength + nx * halfWidth,

               y + uy * pinLength + ny * halfWidth);

    ctx.lineTo(x + ux * pinLength - nx * halfWidth,

               y + uy * pinLength - ny * halfWidth);

    ctx.lineTo(x + ux * taper - nx * halfWidth,

               y + uy * taper - ny * halfWidth);

    ctx.closePath();

    ctx.fill();

    ctx.shadowBlur = 0;

    ctx.fillStyle = "#f2ffff";

    ctx.globalAlpha = alpha * 0.75;

    ctx.beginPath();

    ctx.arc(x + ux * (pinLength - 3), y + uy * (pinLength - 3),

            1.4, 0, TAU);

    ctx.fill();

    ctx.restore();

  }

  function drawBackdrop() {

    ctx.fillStyle = "#0d1017";

    ctx.fillRect(0, 0, width, height);

    const glow = ctx.createRadialGradient(

      cx, cy, radius * 0.3, cx, cy, Math.max(width, height) * 0.7

    );

    glow.addColorStop(0, "#142837");

    glow.addColorStop(0.43, "#111925");

    glow.addColorStop(1, "#0d1017");

    ctx.fillStyle = glow;

    ctx.fillRect(0, 0, width, height);

    ctx.fillStyle = "#7890ae";

    ctx.globalAlpha = 0.10;

    for (let x = 20; x < width; x += 30) {

      for (let y = 20; y < height; y += 30) {

        ctx.beginPath();

        ctx.arc(x, y, 0.65, 0, TAU);

        ctx.fill();

      }

    }

    ctx.globalAlpha = 1;

  }

  function drawWheel() {

    const outerRing = radius + pinLength + 13;

    ctx.save();

    ctx.strokeStyle = "#ffffff06";

    ctx.lineWidth = 1;

    ctx.beginPath();

    ctx.arc(cx, cy, outerRing, 0, TAU);

    ctx.stroke();

    // Tiny orbit markers make rotation readable even before the first hit.

    ctx.strokeStyle = "#7892b821";

    for (let i = 0; i < 32; i++) {

      const a = i / 32 * TAU;

      const inner = outerRing + 5;

      const outer = inner + (i % 4 === 0 ? 5 : 2);

      ctx.beginPath();

      ctx.moveTo(cx + Math.cos(a) * inner, cy + Math.sin(a) * inner);

      ctx.lineTo(cx + Math.cos(a) * outer, cy + Math.sin(a) * outer);

      ctx.stroke();

    }

    for (const pin of pins) {

      const a = pin.localAngle + angle;

      drawPin(

        cx + Math.cos(a) * radius,

        cy + Math.sin(a) * radius,

        a, pin.color

      );

    }

    const fill = ctx.createRadialGradient(

      cx - radius * 0.25, cy - radius * 0.35, 0,

      cx, cy, radius * 1.2

    );

    fill.addColorStop(0, hitFlash > 0.1 ? "#294c5c" : "#233044");

    fill.addColorStop(0.65, "#161e2c");

    fill.addColorStop(1, "#101620");

    ctx.shadowColor = hitFlash > 0.1 ? "#00f2fe" : "#00b9d0";

    ctx.shadowBlur = 15 + hitFlash * 25;

    ctx.fillStyle = fill;

    ctx.beginPath();

    ctx.arc(cx, cy, radius, 0, TAU);

    ctx.fill();

    ctx.lineWidth = 2;

    ctx.strokeStyle = hitFlash > 0.1 ? "#9effff" : "#00dbe7";

    ctx.stroke();

    ctx.shadowBlur = 0;

    ctx.strokeStyle = "#ffffff08";

    ctx.lineWidth = 1;

    ctx.beginPath();

    ctx.arc(cx, cy, radius - 7, 0, TAU);

    ctx.stroke();

    ctx.save();

    ctx.translate(cx, cy);

    ctx.rotate(angle);

    ctx.strokeStyle = "#00f2fe99";

    ctx.lineWidth = 3;

    ctx.lineCap = "round";

    for (let i = 0; i < 3; i++) {

      const start = i / 3 * TAU;

      ctx.beginPath();

      ctx.arc(0, 0, radius - 12, start, start + 0.45);

      ctx.stroke();

    }

    ctx.restore();

    ctx.textAlign = "center";

    ctx.textBaseline = "middle";

    ctx.fillStyle = "#ecf8ff";

    ctx.font = `850 ${Math.max(16, radius * 0.48)}px system-ui, sans-serif`;

    ctx.fillText(remaining, cx, cy - radius * 0.045);

    if (radius > 30) {

      ctx.fillStyle = "#7f92ac";

      ctx.font = `700 ${Math.max(7, radius * 0.105)}px system-ui, sans-serif`;

      ctx.fillText("PINS LEFT", cx, cy + radius * 0.32);

    }

    ctx.restore();

  }

  function drawLauncher() {

    const targetY = cy + radius + pinLength + 20;

    const startY = launchY - 17;

    if (startY > targetY && state === "playing") {

      ctx.save();

      ctx.setLineDash([2, 9]);

      ctx.strokeStyle = "#00f2fe19";

      ctx.lineWidth = 1;

      ctx.beginPath();

      ctx.moveTo(cx, startY);

      ctx.lineTo(cx, targetY);

      ctx.stroke();

      ctx.restore();

    }

    if (["ready", "playing", "paused"].includes(state) && !projectile) {

      const pulse = reducedMotion ? 1 : 0.75 + Math.sin(visualTime * 3.8) * 0.2;

      ctx.save();

      ctx.globalAlpha = pulse;

      ctx.strokeStyle = "#00f2fe35";

      ctx.lineWidth = 1;

      ctx.beginPath();

      ctx.ellipse(cx, launchY + pinLength + 6, 15, 4, 0, 0, TAU);

      ctx.stroke();

      ctx.restore();

      drawPin(cx, launchY, BOTTOM, "#baffff");

    }

    if (projectile) {

      ctx.save();

      const trail = ctx.createLinearGradient(

        cx, projectile.y, cx, projectile.y + pinLength + 45

      );

      trail.addColorStop(0, "#00f2fe45");

      trail.addColorStop(1, "#00f2fe00");

      ctx.fillStyle = trail;

      ctx.fillRect(cx - 2, projectile.y + 5, 4, pinLength + 45);

      ctx.restore();

      drawPin(cx, projectile.y, BOTTOM, "#d0ffff");

    }

    if (brokenPin && ["dying", "over"].includes(state)) {

      drawPin(

        brokenPin.x, brokenPin.y,

        BOTTOM + brokenPin.rotation, "#ff6b91",

        state === "over" ? 0.3 : 1

      );

    }

  }

function drawEffects() {

    ctx.save();

    for (const p of particles) {

      ctx.globalAlpha = clamp(p.life / p.maxLife, 0, 1);

      ctx.fillStyle = p.color;

      ctx.beginPath();

      ctx.arc(p.x, p.y, p.size * (0.4 + 0.6 * p.life / p.maxLife), 0, TAU);

      ctx.fill();

    }

    ctx.textAlign = "left";

    ctx.textBaseline = "middle";

    ctx.font = "800 19px system-ui, sans-serif";

    for (const p of popups) {

      ctx.globalAlpha = clamp(p.life / 0.18, 0, 1);

      ctx.fillStyle = p.color;

      ctx.shadowColor = p.color;

      ctx.shadowBlur = 10;

      ctx.fillText(p.text, p.x, p.y);

    }

    ctx.restore();

  }

  function draw() {

    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);

    ctx.globalAlpha = 1;

    ctx.shadowBlur = 0;

    drawBackdrop();

    ctx.save();

    if (shake > 0 && !reducedMotion) {

      ctx.translate(random(-7, 7) * shake, random(-7, 7) * shake);

    }

    drawWheel();

    drawLauncher();

    drawEffects();

    ctx.restore();

    if (state === "dying") {

      ctx.fillStyle = `rgba(255, 55, 100, ${Math.max(0, 0.13 - stateTime * 0.35)})`;

      ctx.fillRect(0, 0, width, height);

    }

  }

  function frame(timestamp) {

    if (!lastTimestamp) lastTimestamp = timestamp;

    const elapsed = Math.min((timestamp - lastTimestamp) / 1000, 0.08);

    lastTimestamp = timestamp;

    if (state !== "paused" && !document.hidden) {

      accumulator += Math.max(0, elapsed);

      while (accumulator >= FIXED_STEP) {

        update(FIXED_STEP);

        accumulator -= FIXED_STEP;

      }

    } else {

      accumulator = 0;

    }

    draw();

    requestAnimationFrame(frame);

  }

  function bindInput() {

    const isControl = target =>

      target instanceof Element && !!target.closest("button, a, input");

    function handlePress(event) {

      if (isControl(event.target)) return;

      if (event.isPrimary === false) return;

      if (event.button !== undefined && event.button !== 0) return;

      if (event.touches && event.touches.length > 1) return;

      if (event.cancelable) event.preventDefault();

      primaryAction();

    }

    if ("PointerEvent" in window) {

      document.addEventListener("pointerdown", handlePress, { passive: false });

      // Cancel browser gestures without dispatching a duplicate game action.

      document.addEventListener("touchstart", event => {

        if (!isControl(event.target) && event.cancelable) event.preventDefault();

      }, { passive: false });

    } else {

      document.addEventListener("touchstart", handlePress, { passive: false });

      document.addEventListener("mousedown", event => {

        if (!("ontouchstart" in window)) handlePress(event);

      }, { passive: false });

    }

    document.addEventListener("contextmenu", event => event.preventDefault());

    document.addEventListener("dragstart", event => event.preventDefault());

    document.addEventListener("keydown", event => {

      if (event.code === "Space") {

        event.preventDefault();

        if (!event.repeat) primaryAction();

      } else if (event.code === "KeyP" || event.code === "Escape") {

        event.preventDefault();

        if (event.repeat) return;

        if (state === "paused") resumeGame();

        else pauseGame();

      } else if (event.code === "KeyM" && !event.repeat) {

        toggleSound();

      } else if (event.code === "KeyR" && state === "over" && !event.repeat) {

        startGame();

      }

    });

    actionButton.addEventListener("click", primaryAction);

    pauseButton.addEventListener("click", () => {

      unlockAudio();

      if (state === "paused") resumeGame();

      else pauseGame();

    });

    soundButton.addEventListener("click", toggleSound);

    document.addEventListener("visibilitychange", () => {

      if (document.hidden) {

        pauseGame();

        if (audio && audio.state === "running") audio.suspend().catch(() => {});

      }

      lastTimestamp = 0;

      accumulator = 0;

    });

    window.addEventListener("blur", pauseGame);

    window.addEventListener("resize", resize);

    if ("ResizeObserver" in window) {

      new ResizeObserver(resize).observe(stage);

    }

    if (window.visualViewport) {

      window.visualViewport.addEventListener("resize", resize);

    }

  }

  function toggleSound() {

    muted = !muted;

    if (muted && audio && master) {

      master.gain.cancelScheduledValues(audio.currentTime);

      master.gain.setTargetAtTime(0, audio.currentTime, 0.015);

    } else {

      unlockAudio();

      if (audio && master) {

        master.gain.cancelScheduledValues(audio.currentTime);

        master.gain.setTargetAtTime(0.5, audio.currentTime, 0.015);

      }

    }

    updateSoundButton();

    savePreferences();

  }

  function init() {

    loadPreferences();

    updateSoundButton();

    resize();

    configureLevel();

    showModal("ready");

    bindInput();

    requestAnimationFrame(frame);

  }

  init();

})();
